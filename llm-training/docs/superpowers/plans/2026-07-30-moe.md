**Goal:** Replace the dense FFN in each `TransformerBlock` with a top-2-of-4 routed
Mixture-of-Experts FFN, load-balanced by a Switch/GShard auxiliary loss, as a
compute-matched controlled A/B against the 002-rope baseline.

**Architecture:** A new `MoEFeedForward` module (`src/moe.py`) holds a linear router and
a `ModuleList` of expert FFNs. It routes each token to its top-2 experts via softmax
gating, computes all experts and masks (compute-all-then-mask), and stashes a
load-balancing aux loss. `TransformerBlock` selects MoE vs dense on `config.use_moe`;
`TinyTransformer.collect_moe_aux_loss()` gathers the per-layer aux losses for the
training loop. `forward()` still returns only logits, so `generate.py` and the 005 eval
harness are unchanged.

**Tech Stack:** PyTorch, pytest. Same toolchain as experiments 001–009.

## Global Constraints

- Controlled A/B: the **only** variable vs 002-rope is dense FFN → MoE FFN. Same corpus,
  same 002 tokenizer loaded verbatim, `d_model=128 / n_layers=4 / n_heads=4`,
  `context_length=256`, `batch_size=32`, `lr=3e-4`, `seed=0`, 3000 steps, FP32.
- **Compute-matched:** each expert has the same `d_ff` as the dense FFN (512). Params are
  deliberately NOT matched (~4× FFN params) — state this in `mission.md`.
- Experts built via `make_linear(config.quantize_linears, ...)`; this run uses
  `quantize_linears=False`. The **router is always plain `nn.Linear`, FP32**.
- Top-2 of 4 experts. Aux-loss coefficient `α = moe_aux_loss_coef = 0.01`; `α=0` must
  reproduce router collapse (a config knob, not a code path).
- `use_moe=False` MUST leave the dense path byte-for-byte identical to today — experiments
  001–009 must be unaffected.
- Eval for the result is 005's 20-fixed-batch harness (`exp(val_loss)`), compared to the
  002-rope baseline **val loss 4.198 / ppl 66.6**.

import torch

from src.moe import MoEFeedForward
from src.transformer import ModelConfig


def _cfg(**overrides) -> ModelConfig:
    base = dict(vocab_size=100, d_model=16, d_ff=32, n_experts=4, top_k=2, use_moe=True)
    base.update(overrides)
    return ModelConfig(**base)


def test_output_shape_is_preserved():
    torch.manual_seed(0)
    moe = MoEFeedForward(_cfg())
    x = torch.randn(2, 5, 16)
    out = moe(x)
    assert out.shape == x.shape


def test_router_selects_exactly_top_k_experts_per_token():
    # After a forward, exactly top_k experts should have nonzero gate weight
    # for each token. The layer must expose the per-token weight matrix used
    # to combine experts as `moe.last_gate_weights` with shape (T, n_experts).
    torch.manual_seed(0)
    moe = MoEFeedForward(_cfg(n_experts=4, top_k=2))
    x = torch.randn(3, 7, 16)
    moe(x)
    w = moe.last_gate_weights  # (T, n_experts), T = 3*7
    nonzero_per_token = (w > 0).sum(dim=-1)
    assert torch.all(nonzero_per_token == 2)


def test_kept_gate_weights_sum_to_one_per_token():
    torch.manual_seed(0)
    moe = MoEFeedForward(_cfg(n_experts=4, top_k=2))
    x = torch.randn(3, 7, 16)
    moe(x)
    w = moe.last_gate_weights  # (T, n_experts)
    row_sums = w.sum(dim=-1)
    assert torch.allclose(row_sums, torch.ones_like(row_sums), atol=1e-5)


def test_unselected_experts_do_not_affect_output():
    # If an expert's weight is zero for every token, perturbing that expert's
    # parameters must not change the output. Pick the globally least-used expert.
    torch.manual_seed(0)
    moe = MoEFeedForward(_cfg(n_experts=4, top_k=1))  # top_k=1 => 3 experts unused per token
    x = torch.randn(1, 4, 16)
    out_before = moe(x).clone()
    w = moe.last_gate_weights  # (T, n_experts)
    dead_expert = int((w > 0).sum(dim=0).argmin())
    assert (w[:, dead_expert] == 0).all()  # confirm it's unused for all tokens
    with torch.no_grad():
        for p in moe.experts[dead_expert].parameters():
            p.add_(100.0)
    out_after = moe(x)
    assert torch.allclose(out_before, out_after, atol=1e-5)
```

import torch
import torch.nn as nn
import torch.nn.functional as F

from src.ternary_quant import make_linear


class MoEFeedForward(nn.Module):
    def __init__(self, config):
        super().__init__()
        # router: nn.Linear(d_model -> n_experts), bias=False, PLAIN nn.Linear (FP32)
        # experts: nn.ModuleList of n_experts FFNs, each identical to the dense FFN:
        #   make_linear(config.quantize_linears, d_model, d_ff, quantize_activations=config.quantize_activations)
        #   nn.GELU()
        #   make_linear(config.quantize_linears, d_ff, d_model, quantize_activations=config.quantize_activations)
        # stash self.n_experts, self.top_k
        # self.last_gate_weights: Tensor | None = None   # set each forward, (T, n_experts)
        ...

    def forward(self, x):
        # x: (B, S, d_model)
        # 1. flatten to (T, d_model), T = B*S
        # 2. gate_logits = router(flat)            -> (T, n_experts)
        # 3. gate_probs  = softmax(gate_logits)    -> (T, n_experts)   [full softmax; keep it, aux needs it next task]
        # 4. topv, topi  = gate_probs.topk(top_k)  -> (T, top_k) each
        # 5. renormalize topv so the kept weights sum to 1 per token
        # 6. build weights (T, n_experts): zeros, then scatter topv into topi columns
        #    -> store as self.last_gate_weights
        # 7. out = sum over experts e of weights[:, e:e+1] * experts[e](flat)   -> (T, d_model)
        # 8. reshape back to (B, S, d_model)
        ...
```

Hints: `torch.zeros_like(gate_probs).scatter_(1, topi, topv)` builds the sparse weight
matrix in one line. Keep the full `gate_probs` around (return it or stash it) — Task 2's
aux loss needs both `gate_probs` and `topi`.

from src.moe import load_balance_loss


def test_balanced_uniform_routing_gives_minimum_aux():
    # N=4 experts, top_k=1, one token per expert, uniform gate probs.
    # f_i = 1/4 (each expert gets one of four tokens), P_i = 1/4.
    # aux = N * sum_i f_i * P_i = 4 * (4 * (1/4)*(1/4)) = 1.0  -> the balanced minimum.
    n_experts = 4
    gate_probs = torch.full((4, n_experts), 0.25)
    topi = torch.tensor([[0], [1], [2], [3]])
    aux = load_balance_loss(gate_probs, topi, n_experts)
    assert aux.shape == ()  # scalar
    assert torch.isclose(aux, torch.tensor(1.0), atol=1e-5)


def test_collapsed_routing_has_higher_aux_than_balanced():
    n_experts = 4
    # Collapsed: every token routed to expert 0, and probs concentrated there.
    collapsed_probs = torch.tensor([[0.7, 0.1, 0.1, 0.1]] * 4)
    collapsed_topi = torch.zeros(4, 1, dtype=torch.long)
    collapsed = load_balance_loss(collapsed_probs, collapsed_topi, n_experts)
    # Balanced reference from the test above.
    balanced = load_balance_loss(torch.full((4, n_experts), 0.25),
                                 torch.tensor([[0], [1], [2], [3]]), n_experts)
    assert collapsed > balanced


def test_aux_is_non_negative_scalar_after_forward():

    moe = MoEFeedForward(_cfg())
    moe(torch.randn(2, 5, 16))
    assert moe.last_aux_loss.shape == ()
    assert moe.last_aux_loss.item() >= 0.0

```python
def load_balance_loss(gate_probs, topi, n_experts):
    # gate_probs: (T, n_experts) full softmax probabilities
    # topi:       (T, top_k)     indices of the chosen experts per token
    #
    # f_i = fraction of tokens that selected expert i in their top-k
    #       hint: build a (T, n_experts) one-hot of `topi` (e.g.
    #       F.one_hot(topi, n_experts).sum(dim=1).clamp(max=1)), then mean over tokens (dim=0)
    # P_i = mean full-softmax prob mass on expert i  ->  gate_probs.mean(dim=0)
    # aux = n_experts * sum_i (f_i * P_i)            ->  returns a scalar tensor
    ...
```

Then in `forward`, after computing `gate_probs` and `topi`:
`self.last_aux_loss = load_balance_loss(gate_probs, topi, self.n_experts)`.

Note on the `f_i` convention: with top_k>1 the one-hot-then-clamp counts each token once per
distinct chosen expert, so `sum_i f_i = top_k`. The balanced-uniform minimum for the test
above uses `top_k=1` (minimum = 1.0). We accept the standard GShard scaling for `top_k=2` in
training; only the *relative* balanced-vs-collapsed behavior matters for correctness.
